cmake_minimum_required(VERSION 3.20)
project(hakone_builtin_id_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
get_filename_component(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(ADDONS "${SOURCE_ROOT}/modules/gui/qt/addons")
set(HOST_ARCHIVE "" CACHE FILEPATH "Pinned coherent host archive.")
if(NOT EXISTS "${HOST_ARCHIVE}")
    message(FATAL_ERROR "Set HOST_ARCHIVE to the verified host archive.")
endif()
add_library(builtin_id_sources OBJECT "${ADDONS}/addon_manager.cpp" "${ADDONS}/addon_worker.cpp"
    "${ADDONS}/addon_network.cpp" "${ADDONS}/addon_manifest.cpp")
set_target_properties(builtin_id_sources PROPERTIES AUTOMOC OFF)
target_include_directories(builtin_id_sources PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(builtin_id_sources PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
target_link_libraries(builtin_id_sources PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
add_executable(test_addon_runtime "${SOURCE_ROOT}/modules/gui/qt/tests/test_addon_runtime.cpp" $<TARGET_OBJECTS:builtin_id_sources>)
target_include_directories(test_addon_runtime PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(test_addon_runtime PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS VLC_ADDON_SOURCE_ROOT="${SOURCE_ROOT}")
target_link_libraries(test_addon_runtime PRIVATE "${HOST_ARCHIVE}" Qt6::Core Qt6::Gui Qt6::Network Qt6::Test Threads::Threads ZLIB::ZLIB)
add_library(subtitle_id_syntax OBJECT "${SOURCE_ROOT}/modules/gui/qt/util/subtitle_source_service.cpp")
set_target_properties(subtitle_id_syntax PROPERTIES AUTOMOC OFF)
target_include_directories(subtitle_id_syntax PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(subtitle_id_syntax PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
target_link_libraries(subtitle_id_syntax PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
enable_testing()
add_test(NAME addon_runtime COMMAND test_addon_runtime)
