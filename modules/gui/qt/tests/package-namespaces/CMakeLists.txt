cmake_minimum_required(VERSION 3.20)
project(hakone_package_namespace_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
get_filename_component(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(HAKONE_HOST_ARCHIVE "" CACHE FILEPATH "Current coherent normal add-on host archive.")
if(NOT EXISTS "${HAKONE_HOST_ARCHIVE}")
    message(FATAL_ERROR "Set HAKONE_HOST_ARCHIVE to a verified current host archive.")
endif()
add_executable(test_addon_package_namespaces
    "${SOURCE_ROOT}/modules/gui/qt/tests/test_addon_packages.cpp"
    "${SOURCE_ROOT}/modules/gui/qt/addons/addon_manifest.cpp")
target_include_directories(test_addon_package_namespaces PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(test_addon_package_namespaces PRIVATE VLC_ADDON_SOURCE_ROOT="${SOURCE_ROOT}"
    QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
target_link_libraries(test_addon_package_namespaces PRIVATE "${HAKONE_HOST_ARCHIVE}"
    Qt6::Core Qt6::Gui Qt6::Network Qt6::Test Threads::Threads ZLIB::ZLIB)
if(WIN32)
    target_link_libraries(test_addon_package_namespaces PRIVATE Advapi32 ws2_32)
elseif(APPLE)
    target_link_libraries(test_addon_package_namespaces PRIVATE "-framework Security")
endif()
option(HAKONE_PACKAGE_NAMESPACE_SANITIZERS "Enable focused package namespace sanitizers." OFF)
if(HAKONE_PACKAGE_NAMESPACE_SANITIZERS)
    target_compile_options(test_addon_package_namespaces PRIVATE
        -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer -O1 -g1)
    target_link_options(test_addon_package_namespaces PRIVATE -fsanitize=address,undefined,float-cast-overflow)
endif()
enable_testing()
add_test(NAME addon_package_namespaces COMMAND test_addon_package_namespaces)
set_tests_properties(addon_package_namespaces PROPERTIES TIMEOUT 120)
