cmake_minimum_required(VERSION 3.20)
project(hakone_discovery_source_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
get_filename_component(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(ADDONS "${SOURCE_ROOT}/modules/gui/qt/addons")
set(TESTS "${SOURCE_ROOT}/modules/gui/qt/tests")
add_executable(addon_discovery_source_process "${TESTS}/fixtures/addon_discovery_source_process.c")
set_target_properties(addon_discovery_source_process PROPERTIES AUTOMOC OFF)
set(NATIVE_ROOT "${SOURCE_ROOT}/build-redesign" CACHE PATH "The configured native build.")
set(HOST_ARCHIVE "" CACHE FILEPATH "The pinned coherent host archive.")
if(NOT EXISTS "${HOST_ARCHIVE}")
 message(FATAL_ERROR "Set HOST_ARCHIVE to the verified host archive.")
endif()
add_library(source_current OBJECT
 "${ADDONS}/addon_manager.cpp" "${ADDONS}/addon_worker.cpp" "${ADDONS}/addon_network.cpp"
 "${ADDONS}/addon_source_consent.cpp" "${ADDONS}/addon_native_source_resolver.cpp"
 "${ADDONS}/addon_queue_transactions.cpp" "${ADDONS}/addon_discovery_service.cpp"
 "${ADDONS}/addon_player_discovery.cpp" "${ADDONS}/addon_stream_proxy_native.cpp")
set_target_properties(source_current PROPERTIES AUTOMOC OFF)
add_library(source_owner OBJECT "${ADDONS}/addon_discovery_source_owner.cpp" "${ADDONS}/addon_discovery_source_owner.hpp"
 "${ADDONS}/addon_native_queue_store.cpp" "${ADDONS}/addon_native_queue_store.hpp")
foreach(target source_current source_owner)
 target_include_directories(${target} PRIVATE "${SOURCE_ROOT}" "${SOURCE_ROOT}/include" "${SOURCE_ROOT}/modules/gui/qt" "${NATIVE_ROOT}" "${NATIVE_ROOT}/include")
 target_compile_definitions(${target} PRIVATE HAVE_CONFIG_H QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
 target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
endforeach()
foreach(suite discovery_source_owner queue_transactions discovery_service native_queue_store)
 add_executable(test_addon_${suite} "${TESTS}/test_addon_${suite}.cpp" $<TARGET_OBJECTS:source_current> $<TARGET_OBJECTS:source_owner>)
 target_include_directories(test_addon_${suite} PRIVATE "${SOURCE_ROOT}" "${SOURCE_ROOT}/include" "${SOURCE_ROOT}/modules/gui/qt" "${NATIVE_ROOT}" "${NATIVE_ROOT}/include")
 target_compile_definitions(test_addon_${suite} PRIVATE HAVE_CONFIG_H QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS VLC_ADDON_SOURCE_ROOT="${SOURCE_ROOT}" VLC_ADDON_NATIVE_BUILD_ROOT="${NATIVE_ROOT}")
 target_link_libraries(test_addon_${suite} PRIVATE "${HOST_ARCHIVE}" Qt6::Core Qt6::Gui Qt6::Network Qt6::Test Threads::Threads ZLIB::ZLIB "${NATIVE_ROOT}/lib/libvlc.so" "${NATIVE_ROOT}/src/libvlccore.so")
endforeach()
target_compile_definitions(test_addon_discovery_source_owner PRIVATE VLC_ADDON_DISCOVERY_PROCESS_FIXTURE="$<TARGET_FILE:addon_discovery_source_process>")
add_dependencies(test_addon_discovery_source_owner addon_discovery_source_process)
enable_testing()
foreach(suite discovery_source_owner queue_transactions discovery_service native_queue_store)
 add_test(NAME addon_${suite} COMMAND test_addon_${suite})
endforeach()
add_executable(native_module_baseline native_module_baseline.cpp)
target_include_directories(native_module_baseline PRIVATE "${SOURCE_ROOT}/include")
target_link_libraries(native_module_baseline PRIVATE Qt6::Core "${NATIVE_ROOT}/lib/libvlc.so")
# This boundary has no native header, build-directory, or Core dependency.
add_library(source_host_boundary OBJECT "${ADDONS}/addon_queue_transactions.cpp"
 "${ADDONS}/addon_discovery_service.cpp" "${ADDONS}/addon_player_discovery.cpp")
set_target_properties(source_host_boundary PROPERTIES AUTOMOC OFF)
target_include_directories(source_host_boundary PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(source_host_boundary PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
target_link_libraries(source_host_boundary PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
add_executable(test_addon_discovery_service_host "${TESTS}/test_addon_discovery_service.cpp" $<TARGET_OBJECTS:source_host_boundary>)
target_include_directories(test_addon_discovery_service_host PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(test_addon_discovery_service_host PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
target_link_libraries(test_addon_discovery_service_host PRIVATE "${HOST_ARCHIVE}" Qt6::Core Qt6::Gui Qt6::Network Qt6::Test Threads::Threads ZLIB::ZLIB)
add_test(NAME addon_discovery_service_host COMMAND test_addon_discovery_service_host)
