cmake_minimum_required(VERSION 3.20)
project(hakone_invocation_guard_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
get_filename_component(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(ADDONS "${SOURCE_ROOT}/modules/gui/qt/addons")
set(HOST_ARCHIVE "" CACHE FILEPATH "Pinned host archive from the coherent app build.")
if(NOT EXISTS "${HOST_ARCHIVE}")
    message(FATAL_ERROR "Set HOST_ARCHIVE to the verified host archive.")
endif()
add_library(invocation_guard_sources OBJECT
    "${ADDONS}/addon_manager.cpp" "${ADDONS}/addon_worker.cpp" "${ADDONS}/addon_network.cpp")
set_target_properties(invocation_guard_sources PROPERTIES AUTOMOC OFF)
target_include_directories(invocation_guard_sources PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
target_compile_definitions(invocation_guard_sources PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS)
target_link_libraries(invocation_guard_sources PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
foreach(name invocation_guard broker runtime)
    add_executable(test_addon_${name} "${SOURCE_ROOT}/modules/gui/qt/tests/test_addon_${name}.cpp" $<TARGET_OBJECTS:invocation_guard_sources>)
    target_include_directories(test_addon_${name} PRIVATE "${SOURCE_ROOT}/modules/gui/qt")
    target_compile_definitions(test_addon_${name} PRIVATE QT_USE_QSTRINGBUILDER QT_NO_CAST_TO_ASCII QT_NO_CAST_FROM_BYTEARRAY QT_STRICT_ITERATORS VLC_ADDON_SOURCE_ROOT="${SOURCE_ROOT}")
    target_link_libraries(test_addon_${name} PRIVATE "${HOST_ARCHIVE}" Qt6::Core Qt6::Gui Qt6::Network Qt6::Test Threads::Threads ZLIB::ZLIB)
endforeach()
enable_testing()
add_test(NAME addon_invocation_guard COMMAND test_addon_invocation_guard)
add_test(NAME addon_broker COMMAND test_addon_broker)
add_test(NAME addon_runtime COMMAND test_addon_runtime)
